\begin{frame}{Động học nghịch (IK)}
  \begin{columns}[t]
      \begin{column}{0.45\textwidth}
          \begin{block}{Biến đổi Tọa độ \& Xử lý đối xứng}
              \vspace{-0.2cm}
              {\scriptsize
              \begin{equation*}
                  x' = z_B, \quad y' = \sigma_W \frac{W}{2} - y_B, \quad z' = \sigma_L \frac{L}{2} + x_B
              \end{equation*}
              }
          \end{block}
          {\footnotesize
          Tọa độ mục tiêu $(x_B, y_B, z_B)$ từ hệ thân robot được tịnh tiến và hoán đổi trục để chuyển sang không gian cục bộ của từng chân. Bảng hệ số được thiết lập nhằm bù đắp sự khác biệt về vị trí gốc tọa độ và chiều quay của các khớp hông.
          }
          \vspace{0.1cm}
          \begin{table}[htbp]
              \centering
              \resizebox{\linewidth}{!}{
              \def\arraystretch{1.3}
              \begin{tabular}{|l|c|c|c|c|}
                  \hline
                  \textbf{Chân} & $\boldsymbol{\sigma_W}$ & $\boldsymbol{\sigma_L}$ & \textbf{Dấu $\boldsymbol{L_1}$ trong $\boldsymbol{\theta_1}$} & \textbf{Dấu $\boldsymbol{\sin\theta_3}$} \\
                  \hline
                  Left-Front  & $+1$ & $-1$ & $-L_1$ & $-1$ \\
                  \hline
                  Left-Behind & $+1$ & $+1$ & $-L_1$ & $-1$ \\
                  \hline
                  Right-Front & $-1$ & $-1$ & $+L_1$ & $+1$ \\
                  \hline
                  Right-Behind & $-1$ & $+1$ & $+L_1$ & $+1$ \\
                  \hline
              \end{tabular}
              }
              \caption{Bảng hệ số biến đổi tọa độ và dấu IK}
          \end{table}
      \end{column}
      \begin{column}{0.55\textwidth}
          \begin{block}{Nghiệm theo phương pháp đại số}
              \vspace{-0.2cm}
              {\scriptsize
              \begin{align*}
                  \theta_1 &= \mathrm{atan2}(-x',\, y') + \mathrm{atan2}\left(-\sqrt{x'^2 + y'^2 - L_1^2},\; \pm L_1\right) \\
                  \theta_3 &= \mathrm{atan2}(\pm\sqrt{1 - \cos^2\theta_3},\, \cos\theta_3) \quad \text{với} \quad \cos\theta_3 = \frac{p_1^2 + p_2^2 - L_2^2 - L_3^2}{2 L_2 L_3} \\
                  \theta_2 &= \mathrm{atan2}(p_2,\, p_1) - \mathrm{atan2}(L_3 \sin\theta_3,\, L_2 + L_3 \cos\theta_3)
              \end{align*}
              }
              {\scriptsize
              Các biến trung gian: $p_1 = x'\cos\theta_1 + y'\sin\theta_1$ và $p_2 = z'$ (trái) hoặc $p_2 = -z'$ (phải). \\
              \textit{Lưu ý: Các dấu $\pm$ trong phương trình $\theta_1, \theta_3$ được lấy trực tiếp từ Bảng 2.}
              }
          \end{block}
          \vspace{0.2cm}
          {\footnotesize
          Vị trí không gian của các chân được ràng buộc chặt chẽ thông qua hệ tọa độ thân $\{B\}$. Hệ phương trình đại số ánh xạ trực tiếp tọa độ mục tiêu từ hệ thân sang hệ tọa độ góc khớp, đảm bảo sự phối hợp đồng bộ và bảo toàn cấu hình động học tổng thể của robot.
          }
      \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Toán học Bộ tạo Dáng đi (Gait Generator)}
  \begin{columns}[T]
      \begin{column}{0.55\textwidth}
          \textbf{Biến pha $\boldsymbol{\phi_i(t)}$} điều phối nhịp điệu tuần hoàn cho từng chân $i$ dựa trên chu kỳ $T$ và độ lệch pha $\phi_{\text{offset}}$:
          \begin{equation*}
              \phi_i(t) = \left( \frac{t}{T} + \phi_{\text{offset}, i} \right) \bmod 1
          \end{equation*}
          
          \vspace{0.4cm}
          \textbf{Hệ số chu kỳ $\boldsymbol{\beta}$ (Duty factor)} thể hiện tỷ lệ thời gian chân chống đất trong một chu kỳ:
          \begin{equation*}
              \beta = \frac{T_{\text{stance}}}{T_{\text{stance}} + T_{\text{swing}}}
          \end{equation*}
          
          \vspace{0.4cm}
          \textbf{Trạng thái động học của chân:}
          \begin{itemize}
              \item $\phi_i(t) < \beta \Rightarrow$ \textbf{Pha chống (Stance phase)}
              \item $\phi_i(t) \ge \beta \Rightarrow$ \textbf{Pha vung (Swing phase)}
          \end{itemize}
      \end{column}
      \begin{column}{0.45\textwidth}
          \begin{figure}

\begin{frame}{Lựa chọn cơ cấu chấp hành}
  \begin{columns}[T]
      \begin{column}{0.65\textwidth}
          \textbf{Tải trọng thiết kế:} Mô hình tính toán dựa trên lực tĩnh. Các gia tốc động lực học được bù trừ bằng hệ số an toàn $\eta=1.6$: $F_N = \frac{Mg}{2} \approx 12.26\,\text{N}$.
          
          \vspace{0.15cm}
          \textbf{Đánh giá mô-men xoắn (Torque) cực đại:}
          \begin{itemize}
              \item \textbf{Khớp gối (Knee):} Thông qua cơ cấu thanh kéo $CB-CD-DE$:
              \begin{equation*}
                  T_{\text{knee}} = \eta F_N L_3 \sin(\alpha) \frac{L_{DE}}{L_{CB}} \approx \mathbf{17.0\,\text{kg}\cdot\text{cm}}
              \end{equation*}
              $\Rightarrow$ Thỏa mãn định mức an toàn của động cơ ST3215-HS ($20\,\text{kg}\cdot\text{cm}$).
              
              \vspace{0.15cm}
              \item \textbf{Khớp vai (Hip Roll):} Tại góc dang chân cực đại $\theta_1 = 30^\circ$: \\
              $T_{\text{roll}} \approx \mathbf{20.9\,\text{kg}\cdot\text{cm}} < 30\,\text{kg}\cdot\text{cm}$ (Định mức ST3215).
              
              \vspace{0.15cm}
              \item \textbf{Khớp hông (Hip Pitch):} Tại góc vươn chân leo dốc $\alpha = 45^\circ$: \\
              $T_{\text{pitch}} \approx \mathbf{23.2\,\text{kg}\cdot\text{cm}} < 30\,\text{kg}\cdot\text{cm}$ (Định mức ST3215).
          \end{itemize}
      \end{column}

              \vspace{-0.2cm}
              \caption{Các bề mặt nhiễu động thực tế}
          \end{figure}
      \end{column}
      
      \begin{column}{0.6\textwidth}
          \begin{alertblock}{Vẻ đẹp của bộ điều khiển PID}
              \small
              Thay vì các thuật toán đắt đỏ (RL, MPC), PID là một giải pháp \textit{vượt thời gian}: cực kỳ thanh lịch, nhẹ về tính toán và thích ứng tuyệt vời với thế giới thực:
              \begin{itemize}
                  \item \textbf{Khâu P (Bản năng):} Phản ứng tức thời với độ nghiêng, tạo ra mô-men bù để giữ trọng tâm (CoM) trong đa giác chân đế.
                  \item \textbf{Khâu I (Trí nhớ):} Lặng lẽ triệt tiêu các sai số tĩnh, giúp robot thích nghi với thiên vị tải trọng hoặc mặt phẳng nghiêng kéo dài.
                  \item \textbf{Khâu D (Dự đoán):} Đóng vai trò như bộ giảm xóc (damping). Khử các dao động gắt (jittering), mang lại chuyển động mượt mà và tự nhiên.
              \end{itemize}
          \end{alertblock}
      \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Phụ lục A2: Thuật toán Nội suy Đa thức bậc 5 (Quintic)}
  \begin{itemize}
      \item Với thời gian chuẩn hóa $\tau = \frac{i}{N-1}$, đa thức vị trí $s(\tau) = 10\tau^3 - 15\tau^4 + 6\tau^5$ đảm bảo đạo hàm bậc 1 (Vận tốc) và bậc 2 (Gia tốc) tại hai biên $\tau=0, \tau=1$ hoàn toàn bằng $0$.
  \end{itemize}
  \vspace{0.2cm}
  \begin{block}{Quỹ đạo Pha Vung (Swing Phase)}
      Cập nhật vị trí tịnh tiến $(X, Y)$ dựa trên $s(\tau)$:
      \begin{equation*}
          X = p_{start,x} + (p_{end,x} - p_{start,x}) \cdot s(\tau)
      \end{equation*}
      Chiều cao $(Z)$ nội suy dạng Parabol đạt cực đại $h$ tại $\tau = 0.5$:
      \begin{equation*}
          Z = \begin{cases}
          p_{start,z} + h \cdot s(2\tau) & \text{nếu } \tau \leq 0.5 \\
          p_{start,z} + h \cdot \big(1 - s(2(\tau - 0.5))\big) & \text{nếu } \tau > 0.5
          \end{cases}
      \end{equation*}
  \end{block}
\end{frame}

\begin{frame}{Phụ lục A3: Phân tích Ổn định Động \& Tĩnh (ZMP vs Support Polygon)}
  \begin{columns}[t]
      \begin{column}{0.48\textwidth}
          \begin{block}{1. Đa giác chân đế (Support Polygon)}
              \begin{itemize}
                  \item \textbf{Dáng Walk (Tĩnh):} Luôn có ít nhất 3 chân tiếp đất tạo thành hình tam giác. Hình chiếu trọng tâm (CoM) luôn nằm bên trong tam giác này $\rightarrow$ Robot cân bằng hoàn toàn tĩnh.
                  \item \textbf{Dáng Trot (Động):} Chỉ có 2 chân đối chéo chạm đất tạo thành một đường thẳng. CoM luôn có xu hướng rơi ra ngoài $\rightarrow$ Bắt buộc phải liên tục chuyển bước để giữ thăng bằng động.
              \end{itemize}
          \end{block}
      \end{column}
      \begin{column}{0.48\textwidth}
          \begin{block}{2. Tiêu chuẩn ZMP (Zero Moment Point)}
              \begin{itemize}
                  \item ZMP là tiêu chuẩn vàng để đánh giá thăng bằng động. Tuy nhiên, để tính chính xác ZMP cần có \textbf{Cảm biến lực 3 trục} ở mỗi bàn chân để đo áp lực tĩnh và ma sát.
                  \item \textbf{Giải pháp thay thế:} Do giới hạn kinh phí không có cảm biến lực, đề tài sử dụng trực tiếp góc nghiêng từ IMU (Roll, Pitch) để ước lượng trạng thái mất cân bằng và phản xạ bù trừ tức thời.
              \end{itemize}
          \end{block}
      \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Phụ lục B2.1: Chi tiết phân tích mô-men khớp Gối (Knee)}
  \begin{columns}
      \begin{column}{0.6\textwidth}
          Mô-men do lực tiếp đất tác dụng lên khớp gối được cân bằng bởi lực truyền dọc thanh liên kết $CD$, tạo ra mô-men tác dụng lên trục servo. Phương trình:
          \begin{equation*}
              T_{\text{knee}} \approx \eta F_N L_3 \sin(\alpha) \frac{L_{DE}}{L_{CB}}
          \end{equation*}
          Trong dáng đi trot, luôn có hai chân tiếp đất, tải trọng cực đại:
          \begin{equation*}
              F_N = \frac{Mg}{2} = \frac{2.5 \times 9.81}{2} = 12.26\,\text{N}
          \end{equation*}
          Thay $L_3 = 0.125\,\text{m}$, $L_{DE} = 0.024\,\text{m}$, $L_{CB} = 0.025\,\text{m}$, $\eta = 1.6$, $\alpha = 45^\circ$:
          \begin{equation*}
              T_{\text{knee}} \approx 1.6 \times 12.26 \times 0.125 \times \sin(45^\circ) \times \frac{0.024}{0.025}
          \end{equation*}
          \begin{equation*}
              T_{\text{knee}} \approx 1.665\,\text{N}\cdot\text{m} \approx \textbf{17.0 kg.cm}
          \end{equation*}
          \vspace{0.2cm}
          \textbf{Kết luận:} $17.0\,\text{kg}\cdot\text{cm} < 20\,\text{kg}\cdot\text{cm}$ (Định mức ST3215-HS). Khớp gối hoạt động an toàn theo thiết kế lý thuyết.
      \end{column}
      \begin{column}{0.4\textwidth}
          \begin{figure}
              \centering
              \includegraphics[width=0.9\linewidth]{hardware/servo_torque_analysis.png}
          \end{figure}
      \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Phụ lục B2.2: Chi tiết phân tích mô-men khớp Vai (Hip Roll)}
  Khoảng cách hiệu dụng từ trục khớp hông đến mặt đất ở góc khuỵu gối $\beta = 90^\circ$:
  \begin{equation*}
      L_{hd} = \sqrt{L_2^2 + L_3^2 - 2L_2L_3\cos(90^\circ)} = \sqrt{0.106^2 + 0.125^2} \approx 0.164\,\text{m}
  \end{equation*}
  Lực tác dụng tổng cộng $\eta F_N = 1.6 \times 12.26 = 19.62\,\text{N}$. 
  Phương trình mô-men theo góc dang chân $\theta_1$:
  \begin{equation*}
      T_{roll} = \eta F_N \big(L_1 \cos(\theta_1) + L_{hd} \sin(\theta_1)\big) = 19.62 \times \big(0.026 \cos(\theta_1) + 0.164 \sin(\theta_1)\big)
  \end{equation*}
  \begin{itemize}
      \item $\theta_1 = 0^\circ$ (Thẳng đứng): $T_{roll} \approx 0.51\,\text{N}\cdot\text{m} \approx \textbf{5.2 kg.cm}$.
      \item $\theta_1 = 15^\circ$ (Dang trung bình): $T_{roll} \approx 1.33\,\text{N}\cdot\text{m} \approx \textbf{13.5 kg.cm}$.
      \item $\theta_1 = 30^\circ$ (Dang rộng cực đại): $T_{roll} \approx 2.05\,\text{N}\cdot\text{m} \approx \textbf{20.9 kg.cm}$.
  \end{itemize}
  \vspace{0.2cm}
  \textbf{Kết luận:} $20.9\,\text{kg}\cdot\text{cm} < 30\,\text{kg}\cdot\text{cm}$ (Định mức ST3215). Khớp vai hoạt động trong vùng an toàn, đảm bảo hiệu năng.
\end{frame}

\begin{frame}{Phụ lục B2.3: Chi tiết phân tích mô-men khớp Hông (Hip Pitch)}
  \begin{columns}
      \begin{column}{0.65\textwidth}
          Mô-men phụ thuộc vào độ vươn xa của chân theo phương ngang. Với góc nghiêng $\alpha$ so với phương thẳng đứng:
          \begin{equation*}
              T_{pitch} = \eta F_N L_{hd} \sin(\alpha)
          \end{equation*}
          \begin{equation*}
              T_{pitch} = 19.62 \times 0.164 \sin(\alpha) = 3.218 \sin(\alpha) \quad \text{(N}\cdot\text{m)}
          \end{equation*}
          Phân tích các quỹ đạo:
          \begin{itemize}
              \item $\alpha = 0^\circ$ (Đứng thẳng): Trọng lượng truyền dọc chân, $T_{pitch} = 0$.
              \item $\alpha = 15^\circ$ (Bước thường): $T_{pitch} \approx 0.83\,\text{N}\cdot\text{m} \approx \textbf{8.5 kg.cm}$.
              \item $\alpha = 45^\circ$ (Vươn dài leo dốc): $T_{pitch} \approx 2.27\,\text{N}\cdot\text{m} \approx \textbf{23.2 kg.cm}$.
          \end{itemize}
          \vspace{0.2cm}
          \textbf{Kết luận:} $23.2\,\text{kg}\cdot\text{cm} < 30\,\text{kg}\cdot\text{cm}$ (Định mức ST3215). Cơ cấu ST3215 đáp ứng hoàn hảo yêu cầu tải động của khớp hông.
      \end{column}
      \begin{column}{0.35\textwidth}
          \begin{figure}
              \centering

          \end{figure}

  \end{columns}
\end{frame}

\begin{frame}{Phụ lục B3: Tốc độ, Độ phân giải \& Xác nhận từ Mã nguồn}
  \begin{columns}[t]
      \begin{column}{0.48\textwidth}
          \begin{block}{1. Kiểm chứng Tốc độ (Speed)}
              Từ code: \texttt{Waypoint(swing=45)}, $\Delta t = 7\,$ms.\\
              Thời gian pha Swing: $T_{sw} = 45 \times 7\,\text{ms} = 315\,\text{ms}$.\\
              Biên độ góc Gối max $\approx 90^\circ$, yêu cầu:
              \begin{center}
                  $t_{req} = 0.094 \times \frac{90}{60} = \textbf{141 ms}$
              \end{center}
              $141\,\text{ms} \ll 315\,\text{ms}$: \textbf{ST3215-HS đáp ứng dư}.
          \end{block}
          \vspace{0.05cm}
          \begin{block}{2. Độ phân giải Encoder}
              Magnetic Encoder $360^\circ$ / 4096 tick:\\
              $\Rightarrow \Delta\theta = \frac{360}{4096} = \textbf{0.088}^\circ/\textbf{tick}$.\\
              Đủ nhỏ để xử lý vòng lặp IK (sai số $< 0.1^\circ$).
          \end{block}
      \end{column}
      \begin{column}{0.48\textwidth}
          \begin{block}{3. Xác nhận Băng thông UART}
              Giao thức 8N1, Baudrate = 1\,Mbps:\\
              Mỗi giao dịch = $(10+8) \times 10\,$bit $\Rightarrow t_{servo} = 180\,\mu\text{s}$.\\
              Mỗi bus 6 servo: $t_{bus} = 1.08\,$ms.\\
              Hai bus song song $\Rightarrow$ tổng $\approx \textbf{1.1 ms}$.\\
              $1.1\,\text{ms} \ll 7\,\text{ms}$ (chu kỳ vòng lặp PID).
          \end{block}
          \vspace{0.05cm}
          \begin{block}{4. Thông số từ Mã nguồn}
              Trích xuất từ \texttt{gaitGenerator.py}:
              \begin{itemize}
                  \item \texttt{RobotLength(L1=26, L2=106, L3=125)}
                  \item \texttt{Waypoint(stance=200, swing=45)}
                  \item $\beta = \frac{200}{200+45} = \textbf{0.816} > 0.5$ {\small (Trot ổn định)}
              \end{itemize}
          \end{block}
      \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Phụ lục B4: Lý do chọn 2 loại Servo \& Phân tích Tốc độ}
  \small
  \begin{columns}[t]
      \begin{column}{0.48\textwidth}
          \begin{block}{1. So sánh 2 biến thể ST3215}
              \def\arraystretch{1.1}
              \centering
              \begin{tabular}{l c c}
                  \toprule
                  & \textbf{HS} & \textbf{Std} \\
                  \midrule
                  Torque & 20\,kg.cm & 30\,kg.cm \\
                  Tốc độ & 0.09\,s/60$^\circ$ & 0.22\,s/60$^\circ$ \\
                  $\omega_{max}$ & 667$^\circ$/s & 273$^\circ$/s \\
                  Vị trí & Khớp Gối & Khớp Hông \\
                  \bottomrule
              \end{tabular}
          \end{block}
          \begin{alertblock}{Tại sao không dùng HS cho toàn bộ khớp?}
              {\footnotesize Khớp Hông chịu toàn bộ tải trọng thân. Với HS ($20\,$kg.cm):\\
              $\text{SF}_{hip} = \frac{20}{16.0} = 1.25$ (biên an toàn không đủ).\\
              Yêu cầu $30\,$kg.cm (Std) $\Rightarrow$ SF $= 1.88$.}
          \end{alertblock}
      \end{column}
      \begin{column}{0.48\textwidth}
          \begin{block}{2. Kiểm chứng tốc độ Gối (Swing)}
              $T_{sw} = 45 \times 7 = 315\,$ms.\\
              Biên độ Gối max $\approx 90^\circ$:\\
              $\omega_{req} = \frac{90^\circ}{0.315\,\text{s}} = 286^\circ/\text{s}$\\[0.1cm]
              ST3215-HS: $667 \gg 286$ $\checkmark$\\
              ST3215-Std: $273 < 286$ $\times$ {\footnotesize (không đáp ứng)}\\
              $\Rightarrow$ Khớp Gối \textbf{bắt buộc} sử dụng biến thể HS.
          \end{block}
          \begin{block}{3. Kiểm chứng tốc độ Hông (Stance)}
              Biên độ Hông mỗi sải $\approx 15^\circ$:\\
              $\omega_{req} = \frac{15^\circ}{0.315} = 47.6^\circ/\text{s}$\\
              ST3215-Std: $273 \gg 47.6$ $\checkmark$\\
              $\Rightarrow$ Std đáp ứng dư tốc độ, ưu tiên mô-men.
          \end{block}
      \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Phụ lục C1: Ưu điểm của ROS 2 và DDS trong điều khiển Robot}
  \begin{itemize}
      \item \textbf{Loại bỏ điểm lỗi đơn (Single Point of Failure):} ROS~2 sử dụng DDS giao tiếp ngang hàng (Peer-to-Peer), không phụ thuộc \texttt{roscore} như ROS~1. Sự cố tại một node không ảnh hưởng đến các node còn lại.
      \item \textbf{Chính sách chất lượng dịch vụ (QoS):} 
      \begin{itemize}
          \item Cấu hình \texttt{Best\_effort} + \texttt{Keep\_last} cho luồng IMU.
          \item Hệ thống chỉ xử lý mẫu phản hồi mới nhất, loại bỏ các gói tin lỗi thời $\rightarrow$ tối thiểu hóa độ trễ (latency) cho vòng lặp PID.
      \end{itemize}
      \item \textbf{Khả năng mở rộng:} Kiến trúc phân tán cho phép tích hợp thêm các module Navigation, SLAM hoặc AI mà không cần tái cấu trúc lõi điều khiển dáng đi.
  \end{itemize}
\end{frame}

\begin{frame}{Phụ lục C2: Phân tích Chuỗi thời gian \& Nút thắt cổ chai (Bottleneck)}
  \begin{block}{1. Ngân sách thời gian mỗi chu kỳ điều khiển ($T_{cycle} = 7\,$ms)}
      {\small
      \def\arraystretch{1.15}
      \centering
      \begin{tabular}{l c r}
          \toprule
          \textbf{Tác vụ} & \textbf{Tính toán} & \textbf{Thời gian} \\
          \midrule
          ROS\,2 Callback + IK & Python interpreter & $\approx 0.5\,$ms \\
          UART TX+RX (2 bus $\parallel$) & $\frac{6 \times 18 \times 10\,\text{bit}}{1\,\text{Mbps}} = 1.08\,$ms & $\approx 1.1\,$ms \\
          Servo phản hồi cơ học & Nội bộ firmware PI & (chồng lấn) \\
          \midrule
          \textbf{Tổng chiếm dụng} & $0.5 + 1.1$ & $\textbf{1.6\,ms}$ \\
          \textbf{Headroom còn lại} & $7.0 - 1.6$ & $\textbf{5.4\,ms}$ ($77\%$) \\
          \bottomrule
      \end{tabular}}
  \end{block}
  \begin{block}{2. Trường hợp 1 bus duy nhất (12 servo nối tiếp)}
      \footnotesize $t_{bus} = 12 \times 180\,\mu\text{s} = 2.16\,$ms. Tổng: $0.5 + 2.16 = 2.66\,$ms $\ll 7\,$ms $\Rightarrow$ \textbf{UART không phải nút thắt cổ chai.}\\
      Nút thắt nằm ở \textbf{tốc độ đáp ứng cơ học của servo} (xem F7). Việc chia 2 bus giảm xung đột gói tin trên đường truyền bán song công, nhưng \textbf{không} cải thiện tốc độ quay servo.
  \end{block}
\end{frame}

\begin{frame}{Phụ lục C3: Ảnh hưởng Non-Real-Time của Jetson lên Servo Chattering}
  \small
  \begin{columns}[t]
      \begin{column}{0.48\textwidth}
          \begin{block}{1. Jitter của Linux Scheduler}
              {\footnotesize Jetson Nano chạy Ubuntu (CFS scheduler), \textbf{không phải RTOS}. Timer yêu cầu $7\,$ms, nhưng:\\
              $T_{actual} = 7 \pm \delta$, \quad $\delta \approx 2\text{--}5\,$ms\\
              Trường hợp xấu nhất:\\
              $T_{max} = 7 + 5 = 12\,$ms ($+71\%$)\\
              $T_{min} = 7 - 3 = 4\,$ms ($-43\%$)}
          \end{block}
          \begin{alertblock}{2. Hậu quả lên Quintic}
              {\footnotesize Đa thức bậc 5 giả định $\Delta t$ đều. Khi $\Delta t$ dao động: $\ddot{s}(t) \neq 0$ tại biên $\Rightarrow$ gia tốc đột biến. Servo nhận burst lệnh gần nhau rồi khoảng trống $\Rightarrow$ \textbf{Chattering}.}
          \end{alertblock}
      \end{column}
      \begin{column}{0.48\textwidth}
          \begin{block}{3. Cơ chế Chattering trong Servo}
              {\footnotesize
              \begin{itemize}
                  \item Firmware servo: vòng PI nội bộ, sai lệch lớn $\rightarrow$ PWM 100\% (công suất tối đa).
                  \item \textbf{Không có giới hạn gia tốc nội bộ} -- kiểm soát gia tốc hoàn toàn phụ thuộc Quintic.
                  \item Jitter phá vỡ giả định $\Delta t$ đều $\Rightarrow$ Quintic mất tính hiệu lực.
              \end{itemize}}
          \end{block}
          \begin{block}{4. Giải pháp tiềm năng}
              {\footnotesize
              \begin{itemize}
                  \item Kernel PREEMPT\_RT ($\delta < 0.1\,$ms).
                  \item MCU trung gian (STM32) làm bộ đệm RT.
                  \item Bù jitter bằng timestamp nội suy.
              \end{itemize}}
          \end{block}
      \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Phụ lục D1: Tại sao chọn PID + IMU thay vì MPC hoặc RL?}
  \begin{itemize}
      \item \textbf{Ràng buộc cơ cấu chấp hành (Actuator):} Hệ thống sử dụng Bus Servo ST3215 điều khiển vị trí (Position Control) do giới hạn ngân sách. MPC và RL đòi hỏi động cơ BLDC điều khiển mô-men (Torque Control) với băng thông cao hơn nhiều lần.
      \item \textbf{Thiếu cảm biến phản hồi lực:} Không trang bị cảm biến lực (Force Sensor) để đo phản lực mặt đất (GRF) hay Camera chiều sâu để nhận thức địa hình $\rightarrow$ thiếu quan trắc đầu vào cho bộ điều khiển phức tạp.
      \item \textbf{Giới hạn năng lực tính toán:} Jetson Nano qua giao tiếp UART không đáp ứng đủ băng thông để giải bài toán tối ưu lồi (Convex Optimization) liên tục của MPC (yêu cầu tần số $> 500\,$Hz).
      \item \textbf{Hiệu quả chi phí:} PID kết hợp IK cho thời gian tính toán vòng lặp $<1\,$ms, đảm bảo ổn định tư thế với chi phí triển khai tối thiểu.
  \end{itemize}
\end{frame}

\begin{frame}{Phụ lục D2: Bộ lọc Trung bình trượt hàm mũ (EMA)}
  \begin{itemize}
      \item Tín hiệu IMU thô chịu ảnh hưởng nặng từ rung động cơ khí sinh ra bởi động cơ.
      \item Bộ lọc EMA đóng vai trò như một bộ lọc thông thấp IIR (Infinite Impulse Response) bậc 1, giúp làm mượt tín hiệu phản hồi.
  \end{itemize}
  \vspace{0.2cm}
  \begin{block}{1. Phương trình sai phân (Miền thời gian)}
      \begin{equation*}
          \theta_{filtered}[k] = \alpha \cdot \theta_{raw}[k] + (1 - \alpha) \cdot \theta_{filtered}[k-1]
      \end{equation*}
  \end{block}
  \vspace{0.2cm}
  \begin{block}{2. Hàm truyền (Miền Z)}
      Chuyển đổi Z-Transform của bộ lọc:
      \begin{equation*}
          H(z) = \frac{\Theta_{filtered}(z)}{\Theta_{raw}(z)} = \frac{\alpha}{1 - (1-\alpha)z^{-1}}
      \end{equation*}
  \end{block}
  {\small \textit{* $\alpha \in [0, 1]$ là hệ số làm mượt (Smoothing factor) được cấu hình tĩnh.}}
\end{frame}

\begin{frame}{Phụ lục D3: Kiến trúc Lọc tín hiệu đầu vào (Tiền xử lý EMA)}

          \vspace*{0pt}
          \begin{block}{1. Nguồn nhiễu cơ học}

\begin{frame}{Phụ lục D5: Chống bão hòa PID (Anti-Windup Back-Calculation)}
  \begin{itemize}
      \item Khi robot hoạt động thời gian dài trên mặt nghiêng, khâu Tích phân ($I_{term}$) sẽ liên tục cộng dồn gây bão hòa vòng kín (Windup).
  \end{itemize}
  \begin{block}{1. Rò rỉ tích phân (Integral Leakage)}
      Ngăn tích phân tăng vô hạn bằng cách áp dụng hệ số suy hao $\gamma \approx 0.99$:
      \begin{equation*}
          I_{term}[k] = \gamma \cdot I_{term}[k-1] + K_i \cdot e[k] \cdot \Delta t
      \end{equation*}
  \end{block}
  \begin{block}{2. Back-calculation}
      Khấu trừ giá trị dư thừa ở vòng lặp hiện tại khi tín hiệu vượt ngưỡng bão hòa Actuator ($u_{sat}$):
      \begin{equation*}
          I_{term\_adj}[k] = I_{term}[k] + K_{aw} \cdot \left( u_{sat}[k] - u[k] \right)
      \end{equation*}
  \end{block}
  {\small \textit{* Giải pháp này giúp thời gian quá độ giảm đáng kể khi thoát khỏi trạng thái bão hòa, từ đó chống lật robot.}}
\end{frame}

\begin{frame}{Phụ lục D7: Phân biệt vai trò của Bộ lọc EMA và LPF}

  \end{itemize}

  \begin{columns}[t]
      \begin{column}{0.48\textwidth}
          \begin{block}{1. Lọc đầu vào: EMA (Tiền xử lý)}

              \begin{itemize}

                  \item \textbf{Mục đích:} Ngăn hiện tượng rung cơ khí (Mechanical chattering), bảo vệ hệ truyền động bánh răng và giảm tổn hao nhiệt.
              \end{itemize}
          \end{block}
      \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Phụ lục D8: Biến đổi Tọa độ Không gian \& Gain Scheduling}
  \small
  \begin{columns}[t]
      \begin{column}{0.48\textwidth}
          \begin{block}{1. Biến đổi Tọa độ (Ma trận Xoay $R$)}
              {\footnotesize Tín hiệu bù $u_{roll}, u_{pitch}$ là dạng vô hướng tương đối. Để áp dụng cho 4 chân $P_{foot} = [x, y, z]^T$:
              \begin{itemize}
                  \item Tọa độ mục tiêu tính toán: $P_{target} = R(-\theta_{comp}, -\phi_{comp}) \cdot P_{nominal}$
                  \item $R$: Ma trận xoay Euler. Nó \textbf{quay ngược} tọa độ bàn chân một góc đối nghịch, giúp bù lại sai lệch nghiêng thân $\rightarrow$ duy trì thân song song trọng trường trước khi giải Động học ngược (IK).
              \end{itemize}}
          \end{block}
      \end{column}
      \begin{column}{0.48\textwidth}
          \begin{block}{2. Gain Scheduling (Bộ định tuyến ma trận $K_{pid}$)}
              {\footnotesize Động lực học các dáng đi khác biệt rất lớn:
              \begin{itemize}
                  \item \textbf{Trot (Chạy bộ):} Động lượng rơi lớn, điểm tựa ít (2 chân) $\rightarrow$ Yêu cầu $K_p$ cao và $K_d$ gắt để phản xạ tức thời chống sập.
                  \item \textbf{Walk (Đi bộ):} Hệ thống tĩnh ổn định (luôn có 3 chân tựa) $\rightarrow$ Yêu cầu $K_p$ thấp, tập trung giảm vọt lố để bước đi êm ái.
                  \item Hệ thống liên tục chuyển đổi ma trận $K_{pid}$ tuỳ thuộc State Machine hiện tại.
              \end{itemize}}
          \end{block}
      \end{column}
  \end{columns}
\end{frame}
