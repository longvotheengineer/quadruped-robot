            \item \textbf{Mô hình:} \textit{Tích hợp dọc (Vertical Integration)} - tự chủ sản xuất từ linh kiện đến thành phẩm để kiểm soát chất lượng tuyệt đối.
            \item \textbf{Tâm lý:} Ỷ lại vào vinh quang cũ, ngủ quên trên chiến thắng.
        \end{itemize}
        
        \column{.48\textwidth}
        {\large\textcolor{hcmutred}{\textbf{\rule{0.2cm}{0.4cm} Khủng hoảng Bất ngờ (1996)}}}
        \begin{itemize}
            \item \textbf{Đối thủ:} Nắm giữ bằng sáng chế công nghệ dải sóng đột phá mới hơn.
            \item \textbf{Nguy cơ:} Đánh mất vị thế Leader xuống làm Follower trong vòng $\le 2$ năm.
            \item \textbf{Ràng buộc:} \textbf{Thời gian} từ "yếu tố xa xỉ" trở thành "nút thắt sống còn".
        \end{itemize}
    \end{columns}

\begin{frame}{Bài học Kinh nghiệm (Lessons Learned)}
    \begin{columns}[c]
        \column{0.5\textwidth}
        \textbf{Vị thế của Người Quản lý (Project Manager)} \\
        PM không thể nghiêng về cái TÔI kỹ thuật, mà phải là "Bức tường dịch thuật" đứng giữa xung đột \textit{Công nghệ} và \textit{Khả năng sống còn Thương mại}.
        
        \vspace{0.3cm}
        \textbf{Triết lý Đánh đổi (Trade-Off)} \\
