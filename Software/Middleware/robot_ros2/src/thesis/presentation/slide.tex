\section{Lý thuyết động học và quy hoạch dáng đi}
% --------------------------------------------------

\begin{frame}{Mô hình động học nghịch (IK)}
  \begin{columns}
      \begin{column}{0.5\textwidth}
          \begin{itemize}
              \item Sử dụng phương pháp hình học giải tích cho cấu trúc 3-DOF: Coxa, Femur, Tibia.
              \item Cho phép nội suy tọa độ không gian làm việc Cartesian $(x,y,z)$ thành các góc khớp $(\theta_1, \theta_2, \theta_3)$ với tốc độ cao, phục vụ thời gian thực.
          \end{itemize}
      \end{column}
      \begin{column}{0.5\textwidth}
          \begin{figure}
              \centering
              \includegraphics[width=\linewidth, height=0.7\textheight, keepaspectratio]{kinematics/model_reference_frame.png}
              \caption{Mô hình mô phỏng (RViz) với hệ tọa độ tham chiếu tại các khớp}
          \end{figure}
      \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Nghiệm hệ phương trình Động học nghịch (IK)}
    Hệ phương trình lượng giác phi tuyến được phân rã thành các bước giải tuần tự, cho ra nghiệm dạng đóng (closed-form) phục vụ tính toán thời gian thực:
    \vspace{0.3cm}
    \begin{block}{Nghiệm góc khớp $(\theta_1, \theta_2, \theta_3)$}
        \begin{align*}
            \theta_1 &= \mathrm{atan2}(-x',\, y') + \mathrm{atan2}\left(-\sqrt{x'^2 + y'^2 - L_1^2},\; \pm L_1\right) \\
            \theta_3 &= \mathrm{atan2}(\sin\theta_3,\, \cos\theta_3) \quad \text{với} \quad \cos\theta_3 = \frac{p_1^2 + p_2^2 - L_2^2 - L_3^2}{2 L_2 L_3} \\
            \theta_2 &= \mathrm{atan2}(p_2,\, p_1) - \mathrm{atan2}(L_3 \sin\theta_3,\, L_2 + L_3 \cos\theta_3)
        \end{align*}
    \end{block}
    \vspace{0.2cm}
    \begin{itemize}
        \item Trong đó $p_1 = x'\cos\theta_1 + y'\sin\theta_1$ và $p_2 = \pm z'$ là các biến trung gian.
        \item Hệ số dấu ($\pm$) phụ thuộc vào sự đối xứng gương giữa chân Trái/Phải để đảm bảo cấu hình khớp đồng nhất.
    \end{itemize}
\end{frame}

\begin{frame}{Lý thuyết Pha dáng đi và Phân bổ Gaits}
  \begin{columns}
      \begin{column}{0.55\textwidth}
          \begin{itemize}
              \item \textbf{Chu kỳ bước:} Gồm \textbf{Pha vung} (Swing - nhấc chân) và \textbf{Pha chống} (Stance - tạo lực đẩy).
              \item \textbf{Hệ số chu kỳ (Duty factor $\beta$):} Thể hiện tỷ lệ thời gian chân chống đất.
              \vspace{-0.2cm}
              \begin{equation*}
                  \beta = \frac{T_{stance}}{T_{stance} + T_{swing}}
              \end{equation*}
              \vspace{-0.2cm}
              \item \textbf{Phân loại dáng đi:} Dựa vào độ lệch pha $\phi$ và $\beta$.
              \begin{itemize}
                  \item \textbf{Trot (Đi chéo):} 2 chân chéo nhấc đồng thời ($\beta \ge 0.5$). Cân bằng động tốt, tốc độ cao.
                  \item \textbf{Walk (Đi bộ):} Tuần tự từng chân ($\beta \ge 0.75$). Luôn có 3 chân bám đất, cực kỳ ổn định tĩnh.
                  \item \textbf{Pace (Đi kiệu):} 2 chân cùng bên nhấc đồng thời.
              \end{itemize}
          \end{itemize}
      \end{column}
      \begin{column}{0.45\textwidth}
          \begin{figure}
              \centering
              \includegraphics[width=0.95\linewidth, height=0.22\textheight, keepaspectratio]{gait_planner/trot_gait_table.png} \\
              \vspace{0.1cm}
              \includegraphics[width=0.95\linewidth, height=0.22\textheight, keepaspectratio]{gait_planner/pace_gait_table.png} \\
              \vspace{0.1cm}

          \end{figure}
      \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Quy hoạch quỹ đạo chân hình chữ D}
  \begin{columns}
      \begin{column}{0.4\textwidth}
          \begin{itemize}
              \item Quỹ đạo mỗi chân được thiết kế theo dạng hình chữ D (D-shape) trong không gian 3D.
              \item Gồm hai pha: Pha chống (Stance) trên mặt đất và Pha vung (Swing) nhấc theo Parabol.
              \item Đảm bảo chân nhấc lên và hạ xuống mượt mà, không va đập.
          \end{itemize}
      \end{column}

          \end{figure}
      \end{column}
  \end{columns}
\end{frame}

\begin{frame}{So sánh quy hoạch Đa thức (Cubic vs Quintic)}
  \begin{columns}
      \begin{column}{0.4\textwidth}
          \begin{itemize}
              \item Thay vì đa thức bậc 3 (Cubic), hệ thống sử dụng \textbf{Đa thức bậc 5 (Quintic)}.
              \item Triệt tiêu vô hạn gia tốc tại các điểm nhấc/hạ chân ($\ddot{s} = 0$).
              \item Giảm sốc cơ học, bảo vệ hệ thống truyền động.

\begin{frame}{Động lực và Giải pháp Điều khiển cân bằng}
  \begin{columns}
      \begin{column}{0.4\textwidth}
          \textbf{Tại sao cần cân bằng chủ động?} \\
          \vspace{0.2cm}
          \small Môi trường thực tế (boong tàu dập dềnh, cầu tre rung lắc, sàn xe di chuyển) và sự dịch chuyển trọng tâm khi đi bộ khiến robot dễ lật ngã nếu chỉ giữ cứng khớp.
          
          \vspace{0.4cm}
          \begin{figure}
              \centering
              \includegraphics[width=0.31\linewidth, height=0.18\textheight, keepaspectratio]{misc/boat_on_waves.png}
              \hfill
              \includegraphics[width=0.31\linewidth, height=0.18\textheight, keepaspectratio]{misc/bamboo_bridge.png}
              \hfill
              \includegraphics[width=0.31\linewidth, height=0.18\textheight, keepaspectratio]{misc/truck_bed.png}
              \vspace{-0.2cm}
              \caption{Các bề mặt nhiễu động thực tế}
          \end{figure}
      \end{column}
      
      \begin{column}{0.6\textwidth}
          \begin{alertblock}{Giải pháp ổn định tư thế (PID + IMU)}
              \small
              Hệ thống xử lý vòng kín (closed-loop) tốc độ cao, tiết kiệm tài nguyên:
              \begin{itemize}
                  \item \textbf{Cảm nhận:} Đọc góc nghiêng (Roll, Pitch) liên tục từ cảm biến IMU BNO055.
                  \item \textbf{Tính toán bù trừ:} PID tính toán lượng phản xạ (Offset) để triệt tiêu dao động thân.
              \end{itemize}
              \vspace{0.2cm}
              \textbf{Chiến lược áp dụng cho 2 trạng thái:}

\begin{frame}{Phụ lục A1: Bảng thông số MD-H (Modified Denavit-Hartenberg)}
  \begin{itemize}
      \item Tham số hóa động học nghịch bằng quy ước MD-H giúp loại bỏ tính kỳ dị và xác định chính xác ma trận chuyển đổi tọa độ các khớp.

\begin{frame}{Phụ lục A2: Thuật toán Nội suy Đa thức bậc 5 (Quintic)}
  \begin{itemize}
      \item Với thời gian chuẩn hóa $\tau = \frac{i}{N-1}$, đa thức vị trí $s(\tau) = 10\tau^3 - 15\tau^4 + 6\tau^5$ đảm bảo đạo hàm bậc 1 (Vận tốc) và bậc 2 (Gia tốc) tại hai biên $\tau=0, \tau=1$ hoàn toàn bằng $0$.
  \end{itemize}
  \vspace{0.2cm}
  \begin{block}{Quỹ đạo Pha Vung (Swing Phase)}
      Cập nhật vị trí tịnh tiến $(X, Y)$ dựa trên $s(\tau)$:
      \begin{equation*}
          X = p_{start,x} + (p_{end,x} - p_{start,x}) \cdot s(\tau)
      \end{equation*}
      Chiều cao $(Z)$ nội suy dạng Parabol đạt cực đại $h$ tại $\tau = 0.5$:
      \begin{equation*}
          Z = \begin{cases}
          p_{start,z} + h \cdot s(2\tau) & \text{nếu } \tau \leq 0.5 \\
          p_{start,z} + h \cdot \big(1 - s(2(\tau - 0.5))\big) & \text{nếu } \tau > 0.5
          \end{cases}
      \end{equation*}
  \end{block}
\end{frame}

\begin{frame}{Phụ lục A3: Phân tích Ổn định Động \& Tĩnh (ZMP vs Support Polygon)}
  \begin{columns}[t]
      \begin{column}{0.48\textwidth}
          \begin{block}{1. Đa giác chân đế (Support Polygon)}
              \begin{itemize}
                  \item \textbf{Dáng Walk (Tĩnh):} Luôn có ít nhất 3 chân tiếp đất tạo thành hình tam giác. Hình chiếu trọng tâm (CoM) luôn nằm bên trong tam giác này $\rightarrow$ Robot cân bằng hoàn toàn tĩnh.
                  \item \textbf{Dáng Trot (Động):} Chỉ có 2 chân đối chéo chạm đất tạo thành một đường thẳng. CoM luôn có xu hướng rơi ra ngoài $\rightarrow$ Bắt buộc phải liên tục chuyển bước để giữ thăng bằng động.
              \end{itemize}
          \end{block}
      \end{column}
      \begin{column}{0.48\textwidth}
          \begin{block}{2. Tiêu chuẩn ZMP (Zero Moment Point)}
              \begin{itemize}
                  \item ZMP là tiêu chuẩn vàng để đánh giá thăng bằng động. Tuy nhiên, để tính chính xác ZMP cần có \textbf{Cảm biến lực 3 trục} ở mỗi bàn chân để đo áp lực tĩnh và ma sát.
                  \item \textbf{Giải pháp thay thế:} Do giới hạn kinh phí không có cảm biến lực, đề tài sử dụng trực tiếp góc nghiêng từ IMU (Roll, Pitch) để ước lượng trạng thái mất cân bằng và phản xạ bù trừ tức thời.
              \end{itemize}
          \end{block}
      \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Phụ lục B1: Phân tích Năng lượng và Giới hạn Dòng điện}
  \begin{itemize}
      \item \textbf{Yêu cầu dòng điện cực đại:} 
      \begin{itemize}
          \item Dòng kẹt rotor (Stall current) mỗi servo ST3215: $2.0\text{--}2.5\,$A.
          \item Trường hợp xấu nhất (12 servo đồng thời chịu tải cực đại): $I_{peak} \approx \textbf{30\,A}$.
      \end{itemize}
      \item \textbf{Nguồn cấp (Power Supply):}
      \begin{itemize}
          \item Nguồn xung công nghiệp (Switching PSU) \textbf{12V -- 40A} (480\,W).
          \item Biên dự trữ dòng (Headroom): $40 - 30 = 10\,$A, đảm bảo điện áp ổn định, không sụt áp dưới tải đỉnh.
      \end{itemize}
      \item \textbf{Cách ly nguồn ngoại vi:} Jetson Nano ($5\text{V}/3\text{A}$) và Bus Servo Adapter được cấp qua các mạch giảm áp Buck Converter độc lập, cách ly nhiễu điện từ (EMI) do động cơ phát sinh.

\begin{frame}{Phụ lục B2.1: Chi tiết phân tích mô-men khớp Gối (Knee)}
  \begin{columns}
      \begin{column}{0.6\textwidth}
          Mô-men do lực tiếp đất tác dụng lên khớp gối được cân bằng bởi lực truyền dọc thanh liên kết $CD$, tạo ra mô-men tác dụng lên trục servo. Phương trình:
          \begin{equation*}
              T_{\text{knee}} \approx \eta F_N L_3 \sin(\alpha) \frac{L_{DE}}{L_{CB}}
          \end{equation*}
          Trong dáng đi trot, luôn có hai chân tiếp đất, tải trọng cực đại:
          \begin{equation*}
              F_N = \frac{Mg}{2} = \frac{2.5 \times 9.81}{2} = 12.26\,\text{N}
          \end{equation*}
          Thay $L_3 = 0.125\,\text{m}$, $L_{DE} = 0.024\,\text{m}$, $L_{CB} = 0.025\,\text{m}$, $\eta = 1.6$, $\alpha = 45^\circ$:
          \begin{equation*}
              T_{\text{knee}} \approx 1.6 \times 12.26 \times 0.125 \times \sin(45^\circ) \times \frac{0.024}{0.025}
          \end{equation*}
          \begin{equation*}
              T_{\text{knee}} \approx 1.665\,\text{N}\cdot\text{m} \approx \textbf{17.0 kg.cm}
          \end{equation*}
          \vspace{0.2cm}
          \textbf{Kết luận:} $17.0\,\text{kg}\cdot\text{cm} < 20\,\text{kg}\cdot\text{cm}$ (Định mức ST3215-HS). Khớp gối hoạt động an toàn theo thiết kế lý thuyết.

\begin{frame}{Phụ lục B2.2: Chi tiết phân tích mô-men khớp Vai (Hip Roll)}
  Khoảng cách hiệu dụng từ trục khớp hông đến mặt đất ở góc khuỵu gối $\beta = 90^\circ$:
  \begin{equation*}
      L_{hd} = \sqrt{L_2^2 + L_3^2 - 2L_2L_3\cos(90^\circ)} = \sqrt{0.106^2 + 0.125^2} \approx 0.164\,\text{m}
  \end{equation*}
  Lực tác dụng tổng cộng $\eta F_N = 1.6 \times 12.26 = 19.62\,\text{N}$. 
  Phương trình mô-men theo góc dang chân $\theta_1$:
  \begin{equation*}
      T_{roll} = \eta F_N \big(L_1 \cos(\theta_1) + L_{hd} \sin(\theta_1)\big) = 19.62 \times \big(0.026 \cos(\theta_1) + 0.164 \sin(\theta_1)\big)
  \end{equation*}
  \begin{itemize}
      \item $\theta_1 = 0^\circ$ (Thẳng đứng): $T_{roll} \approx 0.51\,\text{N}\cdot\text{m} \approx \textbf{5.2 kg.cm}$.
      \item $\theta_1 = 15^\circ$ (Dang trung bình): $T_{roll} \approx 1.33\,\text{N}\cdot\text{m} \approx \textbf{13.5 kg.cm}$.
      \item $\theta_1 = 30^\circ$ (Dang rộng cực đại): $T_{roll} \approx 2.05\,\text{N}\cdot\text{m} \approx \textbf{20.9 kg.cm}$.
  \end{itemize}
  \vspace{0.2cm}
  \textbf{Kết luận:} $20.9\,\text{kg}\cdot\text{cm} < 30\,\text{kg}\cdot\text{cm}$ (Định mức ST3215). Khớp vai hoạt động trong vùng an toàn, đảm bảo hiệu năng.
\end{frame}

\begin{frame}{Phụ lục C3: Ảnh hưởng Non-Real-Time của Jetson lên Servo Chattering}
  \small
  \begin{columns}[t]
      \begin{column}{0.48\textwidth}
          \begin{block}{1. Jitter của Linux Scheduler}
              {\footnotesize Jetson Nano chạy Ubuntu (CFS scheduler), \textbf{không phải RTOS}. Timer yêu cầu $7\,$ms, nhưng:\\
              $T_{actual} = 7 \pm \delta$, \quad $\delta \approx 2\text{--}5\,$ms\\
              Trường hợp xấu nhất:\\
              $T_{max} = 7 + 5 = 12\,$ms ($+71\%$)\\
              $T_{min} = 7 - 3 = 4\,$ms ($-43\%$)}
          \end{block}
          \begin{alertblock}{2. Hậu quả lên Quintic}
              {\footnotesize Đa thức bậc 5 giả định $\Delta t$ đều. Khi $\Delta t$ dao động: $\ddot{s}(t) \neq 0$ tại biên $\Rightarrow$ gia tốc đột biến. Servo nhận burst lệnh gần nhau rồi khoảng trống $\Rightarrow$ \textbf{Chattering}.}
          \end{alertblock}
      \end{column}
      \begin{column}{0.48\textwidth}
          \begin{block}{3. Cơ chế Chattering trong Servo}
              {\footnotesize
              \begin{itemize}
                  \item Firmware servo: vòng PI nội bộ, sai lệch lớn $\rightarrow$ PWM 100\% (công suất tối đa).
                  \item \textbf{Không có giới hạn gia tốc nội bộ} -- kiểm soát gia tốc hoàn toàn phụ thuộc Quintic.
                  \item Jitter phá vỡ giả định $\Delta t$ đều $\Rightarrow$ Quintic mất tính hiệu lực.
              \end{itemize}}
          \end{block}
          \begin{block}{4. Giải pháp tiềm năng}
              {\footnotesize
              \begin{itemize}
                  \item Kernel PREEMPT\_RT ($\delta < 0.1\,$ms).
                  \item MCU trung gian (STM32) làm bộ đệm RT.
                  \item Bù jitter bằng timestamp nội suy.
              \end{itemize}}
          \end{block}
      \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Phụ lục D1: Tại sao chọn PID + IMU thay vì MPC hoặc RL?}
  \begin{itemize}
      \item \textbf{Ràng buộc cơ cấu chấp hành (Actuator):} Hệ thống sử dụng Bus Servo ST3215 điều khiển vị trí (Position Control) do giới hạn ngân sách. MPC và RL đòi hỏi động cơ BLDC điều khiển mô-men (Torque Control) với băng thông cao hơn nhiều lần.
      \item \textbf{Thiếu cảm biến phản hồi lực:} Không trang bị cảm biến lực (Force Sensor) để đo phản lực mặt đất (GRF) hay Camera chiều sâu để nhận thức địa hình $\rightarrow$ thiếu quan trắc đầu vào cho bộ điều khiển phức tạp.
      \item \textbf{Giới hạn năng lực tính toán:} Jetson Nano qua giao tiếp UART không đáp ứng đủ băng thông để giải bài toán tối ưu lồi (Convex Optimization) liên tục của MPC (yêu cầu tần số $> 500\,$Hz).
      \item \textbf{Hiệu quả chi phí:} PID kết hợp IK cho thời gian tính toán vòng lặp $<1\,$ms, đảm bảo ổn định tư thế với chi phí triển khai tối thiểu.
  \end{itemize}
\end{frame}

\begin{frame}{Phụ lục D2: Bộ lọc Trung bình trượt hàm mũ (EMA)}
  \begin{itemize}
      \item Tín hiệu IMU thô chịu ảnh hưởng nặng từ rung động cơ khí sinh ra bởi động cơ.
      \item Bộ lọc EMA đóng vai trò như một bộ lọc thông thấp IIR (Infinite Impulse Response) bậc 1, giúp làm mượt tín hiệu phản hồi.
  \end{itemize}
  \vspace{0.2cm}
  \begin{block}{1. Phương trình sai phân (Miền thời gian)}
      \begin{equation*}
          \theta_{filtered}[k] = \alpha \cdot \theta_{raw}[k] + (1 - \alpha) \cdot \theta_{filtered}[k-1]
      \end{equation*}
  \end{block}
  \vspace{0.2cm}
  \begin{block}{2. Hàm truyền (Miền Z)}
      Chuyển đổi Z-Transform của bộ lọc:
      \begin{equation*}
          H(z) = \frac{\Theta_{filtered}(z)}{\Theta_{raw}(z)} = \frac{\alpha}{1 - (1-\alpha)z^{-1}}
      \end{equation*}
  \end{block}
  {\small \textit{* $\alpha \in [0, 1]$ là hệ số làm mượt (Smoothing factor) được cấu hình tĩnh.}}
\end{frame}

\begin{frame}{Phụ lục D5: Chống bão hòa PID (Anti-Windup Back-Calculation)}
  \begin{itemize}
      \item Khi robot hoạt động thời gian dài trên mặt nghiêng, khâu Tích phân ($I_{term}$) sẽ liên tục cộng dồn gây bão hòa vòng kín (Windup).
  \end{itemize}
  \begin{block}{1. Rò rỉ tích phân (Integral Leakage)}
      Ngăn tích phân tăng vô hạn bằng cách áp dụng hệ số suy hao $\gamma \approx 0.99$:
      \begin{equation*}
          I_{term}[k] = \gamma \cdot I_{term}[k-1] + K_i \cdot e[k] \cdot \Delta t
      \end{equation*}
  \end{block}
  \begin{block}{2. Back-calculation}
      Khấu trừ giá trị dư thừa ở vòng lặp hiện tại khi tín hiệu vượt ngưỡng bão hòa Actuator ($u_{sat}$):
      \begin{equation*}
          I_{term\_adj}[k] = I_{term}[k] + K_{aw} \cdot \left( u_{sat}[k] - u[k] \right)
      \end{equation*}
  \end{block}
  {\small \textit{* Giải pháp này giúp thời gian quá độ giảm đáng kể khi thoát khỏi trạng thái bão hòa, từ đó chống lật robot.}}
\end{frame}

\begin{frame}{Phụ lục D6: Bộ lọc ngõ ra (LPF), Bão hòa \& Feed-Forward}
  \begin{columns}[T]
      \begin{column}{0.55\textwidth}
          \vspace*{0pt}
          \begin{block}{1. Cắt cực đại (Saturation)}
              {\scriptsize Giới hạn tín hiệu bù $u(t) \in [\pm Max\_Comp]$ để ngăn lệnh xuất đột biến gây ngã.\\
              \textcolor{red}{\textbf{Nhược điểm:}} Giảm năng lực phục hồi trên mặt dốc lớn.}
          \end{block}
          \begin{block}{2. Bộ lọc thông thấp (LPF)}
              {\scriptsize Làm mượt tín hiệu điều khiển sau PID, tránh gây giật cục và sốc cơ khí.\\
              \textcolor{red}{\textbf{Nhược điểm:}} Gây trễ pha, giảm biên dự trữ ổn định và phản ứng chậm.}
          \end{block}
          \begin{block}{3. Tín hiệu định hướng (Feed-Forward)}
              {\scriptsize Cộng thêm bù trước Feed-Forward (\textcolor{green}{xanh lá}) để triệt tiêu độ nghiêng tự nhiên của dáng đi Trot.\\
              \textcolor{red}{\textbf{Nhược điểm:}} Là vòng hở. Nếu ước lượng mô hình sai lệch, FF sẽ bù sai.}
          \end{block}

\begin{frame}{Phụ lục D7: Phân biệt vai trò của Bộ lọc EMA và LPF}
  \begin{itemize}
      \item Trong sơ đồ kiến trúc PID 5 giai đoạn, hệ thống sử dụng hai bộ lọc ở 2 khâu với mục đích hoàn toàn độc lập:
  \end{itemize}
  \vspace{0.2cm}
  \begin{columns}[t]
      \begin{column}{0.48\textwidth}
          \begin{block}{1. Lọc đầu vào: EMA (Tiền xử lý)}
              \begin{itemize}
                  \item \textbf{Vị trí:} Xử lý dữ liệu thô (Raw Data) ngay sau khi đọc từ cảm biến IMU (BNO055).
                  \item \textbf{Vai trò:} Khử nhiễu tần số cao (High-frequency noise) do rung động cơ học truyền lên khung robot.
                  \item \textbf{Đặc điểm:} Thuật toán EMA (Exponential Moving Average) -- độ phức tạp $O(1)$, tiết kiệm bộ nhớ.
              \end{itemize}
          \end{block}
      \end{column}
      \begin{column}{0.48\textwidth}
          \begin{block}{2. Lọc đầu ra: LPF (Hậu xử lý)}
              \begin{itemize}
                  \item \textbf{Vị trí:} Nằm ở khâu cuối cùng, làm mượt tín hiệu bù ($u$) trước khi cộng vào biên độ góc gửi xuống Servo.
                  \item \textbf{Vai trò:} Triệt tiêu xung nhiễu (Spikes) từ khâu Vi phân (D-term).
                  \item \textbf{Mục đích:} Ngăn hiện tượng rung cơ khí (Mechanical chattering), bảo vệ hệ truyền động bánh răng và giảm tổn hao nhiệt.
              \end{itemize}
          \end{block}
      \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Phụ lục D8: Biến đổi Tọa độ Không gian \& Gain Scheduling}
  \small
  \begin{columns}[t]
      \begin{column}{0.48\textwidth}
          \begin{block}{1. Biến đổi Tọa độ (Ma trận Xoay $R$)}
              {\footnotesize Tín hiệu bù $u_{roll}, u_{pitch}$ là dạng vô hướng tương đối. Để áp dụng cho 4 chân $P_{foot} = [x, y, z]^T$:
              \begin{itemize}
                  \item Tọa độ mục tiêu tính toán: $P_{target} = R(-\theta_{comp}, -\phi_{comp}) \cdot P_{nominal}$
                  \item $R$: Ma trận xoay Euler. Nó \textbf{quay ngược} tọa độ bàn chân một góc đối nghịch, giúp bù lại sai lệch nghiêng thân $\rightarrow$ duy trì thân song song trọng trường trước khi giải Động học ngược (IK).
              \end{itemize}}
          \end{block}
      \end{column}
      \begin{column}{0.48\textwidth}
          \begin{block}{2. Gain Scheduling (Bộ định tuyến ma trận $K_{pid}$)}
              {\footnotesize Động lực học các dáng đi khác biệt rất lớn:
              \begin{itemize}
                  \item \textbf{Trot (Chạy bộ):} Động lượng rơi lớn, điểm tựa ít (2 chân) $\rightarrow$ Yêu cầu $K_p$ cao và $K_d$ gắt để phản xạ tức thời chống sập.
                  \item \textbf{Walk (Đi bộ):} Hệ thống tĩnh ổn định (luôn có 3 chân tựa) $\rightarrow$ Yêu cầu $K_p$ thấp, tập trung giảm vọt lố để bước đi êm ái.
                  \item Hệ thống liên tục chuyển đổi ma trận $K_{pid}$ tuỳ thuộc State Machine hiện tại.
              \end{itemize}}
          \end{block}
      \end{column}
  \end{columns}
\end{frame}



\begin{frame}{Phụ lục E1: Phân tích Nguyên nhân "Kéo lê chân" trên thực tế}
  \begin{block}{1. Sai lệch khối lượng thiết kế so với thực tế}
      Lý thuyết $M = 2.5\,\text{kg}$. Thực tế gia công in 3D (in-fill đặc), thêm phụ kiện ốc vít và tản nhiệt làm nguyên mẫu nặng lên $\textbf{3.0 kg}$ (+ $20\%$). Lực ép mỗi chân (Trot): $F_N = 3.0 / 2 = 1.5\,\text{kg}$.
  \end{block}
  \vspace{0.05cm}
  \begin{columns}[t]
      \begin{column}{0.48\textwidth}
          \begin{alertblock}{2. Quá tải Khớp Gối (Knee)}
              Servo ST3215-HS (định mức $20\,\text{kg.cm}$).\\
              Mô-men thực tế ($\eta=1.6$):\\
              $T_{knee} = 1.6 \times 1.5 \times 12.5 \times \sin45^\circ \times \frac{24}{25}$\\
              $\approx \textbf{20.4 kg.cm} > 20\,\text{kg.cm}$\\
              $\Rightarrow$ Hệ số an toàn: $\frac{20}{20.4} \approx \textbf{0.98} < 1$.\\
              \textit{Hậu quả:} Động cơ gối \textbf{Overload}, không thể nâng chân đạt Lift height, gây hiện tượng kéo lê.
          \end{alertblock}
      \end{column}
      \begin{column}{0.48\textwidth}
          \begin{alertblock}{3. Giảm biên độ Khớp Hông (Hip)}
              Servo ST3215 (định mức $30\,\text{kg.cm}$).\\
              Mô-men thực tế ($\alpha=45^\circ$):\\
              $T_{pitch} = 1.6 \times 1.5 \times 16.4 \times \sin45^\circ$\\
              $\approx \textbf{27.8 kg.cm}$\\
              $\Rightarrow$ Hệ số an toàn: $\frac{30}{27.8} \approx \textbf{1.08}$.\\
              \textit{Hậu quả:} Tải tĩnh sát ngưỡng bão hòa sinh nhiệt cao. Vòng lặp PID nội bộ phản hồi chậm $\Rightarrow$ mất bước, lết gót.
          \end{alertblock}
      \end{column}
  \end{columns}
\end{frame}
