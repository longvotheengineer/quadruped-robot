\section{1. Giới thiệu đề tài}
% =============================================

\begin{frame}{1. Giới thiệu đề tài}
    \textbf{Tầm quan trọng của Robot bốn chân (Quadruped Robot):}
    \begin{itemize}
        \item Khả năng di chuyển linh hoạt trên các địa hình phức tạp, gồ ghề (nơi robot dùng bánh xe bị hạn chế).
        \item Được ứng dụng rộng rãi trong thám hiểm, cứu hộ cứu nạn, kiểm tra công nghiệp.
    \end{itemize}
    \vspace{0.3cm}
    \textbf{Mục tiêu Đồ án:}
    \begin{itemize}
        \item Xây dựng một mô hình robot bốn chân hoàn chỉnh (kích thước để bàn) với 12 bậc tự do (DOF).
        \item Thiết kế và triển khai thuật toán giải bài toán động học nghịch (Inverse Kinematics).
        \item Xây dựng hệ thống điều khiển thăng bằng (Balance Control) và tạo quỹ đạo dáng đi linh hoạt dựa trên dữ liệu phản hồi từ cảm biến IMU, triển khai trên nền tảng ROS 2.
    \end{itemize}
\end{frame}

\section{2. Cơ sở lý thuyết}
% =============================================

\begin{frame}{2. Cấu trúc Động học (Kinematics)}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Động học nghịch (Inverse Kinematics):}
            \begin{itemize}
                \item Tính toán góc các khớp ($\theta_1, \theta_2, \theta_3$) dựa trên tọa độ vị trí mong muốn của bàn chân $(x,y,z)$.
                \item Phương pháp biến đổi tọa độ Denavit-Hartenberg (DH).
            \end{itemize}
            \vspace{0.2cm}
            \begin{block}{Công thức giải IK (ví dụ khớp Hip Roll)}
                $$ \theta_1 = -\arctan \left( \frac{y}{z} \right) - \arctan \left( \frac{L_1}{\sqrt{y^2 + z^2 - L_1^2}} \right) $$
            \end{block}
        \end{column}
        \begin{column}{0.5\textwidth}
            \centering
            \includegraphics[height=0.65\textheight]{fig_model_referenceframe.png}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{2. Quy hoạch Quỹ đạo: Đa thức Bậc 5 (Quintic)}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            Để đảm bảo chân robot bước đi êm ái, vận tốc và gia tốc liên tục, không gây giật cục động cơ, nhóm sử dụng **Đa thức bậc 5 (Quintic Polynomial)**:
            $$ y(t) = a_0 + a_1 t + a_2 t^2 + a_3 t^3 + a_4 t^4 + a_5 t^5 $$
            \vspace{0.2cm}
            \begin{itemize}
                \item Đạt được sự trơn tru không chỉ ở \textbf{vận tốc} mà còn ở \textbf{gia tốc} tại các điểm biên (lúc nhấc chân và chạm đất).
                \item Vượt trội hơn hẳn so với nội suy bậc 3 (Cubic) thường có gia tốc bị gãy khúc, gây rung lắc cơ khí.
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \centering
            \includegraphics[width=\textwidth]{fig_cubic_vs_quintic.pdf}
            \footnotesize{So sánh gia tốc Cubic và Quintic}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{2. Thiết kế Dáng đi (Gait Generation)}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Dáng đi Trot (Chạy kiệu):}
            \begin{itemize}
                \item Là dáng đi phổ biến nhất, mang lại sự cân bằng hoàn hảo giữa tốc độ và độ ổn định.
                \item Chân vận động theo cặp chéo nhau: LF (Trái-Trước) và RB (Phải-Sau) di chuyển cùng lúc, xen kẽ với cặp RF và LB.
                \item Sử dụng hệ số chu kỳ $\beta = 0.5$.
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \centering
            \includegraphics[width=\textwidth]{fig22-Trot gait timing diagram β = 0.5 .png}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{2. Điều khiển Thăng bằng (Balance Control)}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Hệ thống Adaptive PID dựa trên IMU:}
            \begin{itemize}
                \item \textbf{Chế độ HOME (Đứng yên):} Cảm biến bù trực tiếp góc Roll/Pitch vào không gian khớp (joint-space). Chống chịu nhiễu tốt khi đứng tại chỗ.
                \item \textbf{Chế độ GAIT (Đang di chuyển):} Khung tọa độ liên tục thay đổi. Tín hiệu bù được tính toán qua ma trận xoay trong không gian làm việc (task-space), kết hợp bộ lọc phân biệt pha (phase-aware filter) để không cản trở quỹ đạo vung chân.
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \centering
            \includegraphics[width=1.0\textwidth]{fig45-Locomotion_Planner_Balance_Controller.png}
        \end{column}
    \end{columns}
\end{frame}

\section{4. Hệ thống Phần mềm (ROS 2)}
% =============================================

\begin{frame}{4. Kiến trúc ROS 2 (Robot Operating System 2)}
    \begin{itemize}
        \item \textbf{Gait Node:} Sinh quỹ đạo theo thời gian thực (Quintic + Timing).
        \item \textbf{Kinematics Node:} Chuyển tọa độ Task-space $(x,y,z)$ thành Joint-space ($\theta$).
        \item \textbf{Balance Node:} PID Controller thu thập IMU (I2C) và sinh tọa độ bù trừ.
        \item \textbf{Serial Node:} Đóng gói dữ liệu gửi qua UART (1 Mbps) tới mạch Adapter điều khiển Servo.
    \end{itemize}

    \includegraphics[width=0.8\textwidth]{fig_ros2_architecture.pdf}
\end{frame}
